39 case CLOD_FILE_READ | CLOD_FILE_DIRECTORY:

41 case CLOD_FILE_READ | CLOD_FILE_WRITE:
42 o_flags |= O_RDWR; break;
43 case CLOD_FILE_READ | CLOD_FILE_WRITE | CLOD_FILE_TRUNCATE:
44 o_flags |= O_RDWR | O_TRUNC; break;
45 case CLOD_FILE_WRITE:
46 o_flags |= O_WRONLY; break;
47 case CLOD_FILE_WRITE | CLOD_FILE_TRUNCATE:
48 o_flags |= O_WRONLY | O_TRUNC; break;
49 case CLOD_FILE_DIRECTORY:
50 o_flags |= O_DIRECTORY; break;
51
52 default:

54 }
55
56 if (flags & CLOD_FILE_CREATE) {
57 o_flags |= O_CREAT;
58 }
59
60 if (flags & CLOD_FILE_APPEND) {
61 o_flags |= O_APPEND;
62 }
63
64 long res;
65 if (directory) {
66 res = syscall(__NR_openat, directory->unix_fd, (long)path, o_flags, 0664);
67 } else {
68 res = syscall(__NR_open, (long)path, o_flags, 0664);

128int clod_file_stream_readdir(clod_stream *self, struct clod_string *dst) {
129 if (dst->cap <= dst->len) {
130 return CLOD_ERR_OK;
131 }
132
133 const long res = syscall(__NR_getdents64,
134 ((clod_file*)self)->unix_fd,
135 (long)(dst->ptr + dst->len),
136 dst->cap - dst->len
137 );
138
139 if (res == 0) {
140 return CLOD_ERR_EOF;
141 }
142
143 if (res == -EBADF) {
144 return CLOD_ERR_INVALID;
145 }
146
147 if (res < 0) {
148 return (int)-res;
149 }
150
151 size_t off = 0;
152 while (off < (size_t)res) {
153 void *ptr = dst->ptr + dst->len + off;
154 const struct linux_dirent ent = *(struct linux_dirent*)ptr;
155 struct clod_dirent *clod_ent = ptr;
156
157 off += ent.size;
158
159 clod_ent->next = off < (size_t)res ? (struct clod_dirent*)((char*)ptr + ent.size) : nullptr;
160 clod_ent->id = (uintptr_t)ent.ino;
161 clod_ent->name_size = ent.size - (unsigned short)offsetof(struct linux_dirent, name) - 1;
162 switch (ent.type) {
163 case DT_DIR: clod_ent->type = CLOD_DIRENT_DIRECTORY; break;
164 case DT_REG: clod_ent->type = CLOD_DIRENT_FILE; break;
165 default: clod_ent->type = ent.type;
166 }
167 }
168 return CLOD_ERR_OK;
169}
170int clod_file_stream_write(clod_stream *self, struct clod_string *src) {
171 if (src->len == 0) {
172 return CLOD_ERR_OK;
173 }
174
175 while (src->len) {
176 const long ret = syscall(__NR_write, ((clod_file*)self)->unix_fd, (long)src->ptr, src->len);
177
178 if (ret > 0) {
179 src->ptr += ret;
180 src->len -= ret;
181 if (src->cap) {
182 src->cap -= ret;
183 }
184 }
185
186 if (ret < 0) {
187 return (int)-ret;
188 }
189 }
