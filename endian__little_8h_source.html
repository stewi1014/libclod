161static inline bool leuv_enc(uint8_t *ptr, const void *end, uint64_t val) {
162 const uint8_t size = leuv_size(val);
163 uint8_t i = 0;
164 while (i + 1 < size) {
165 if (ptr + i == end) return false;
166 ptr[i] = (uint8_t)(val >> (i * 7) | 0b10000000);
167 i++;
168 }
169 if (ptr + i == end) return false;
170 ptr[i] = (uint8_t)(val >> (i * 7) & 0b01111111);
171 return true;
172}
