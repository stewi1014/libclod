161static inline bool beuv_enc(uint8_t *ptr, const void *end, uint64_t val) {
162 const uint8_t size = beuv_size(val);
163 uint8_t i = 0;
164 while (i + 1 < size) {
165 if (ptr + i == end) return false;
166 ptr[i] = (uint8_t)(val >> ((size - i - 1) * 7) | 0b10000000);
167 i++;
168 }
169 if (ptr + i == end) return false;
170 ptr[i] = (uint8_t)(val & 0b01111111);
171 return true;
172}

223static inline bool beiv_dec(const uint8_t *ptr, const void *end, int64_t *val) {
224 int64_t ret = 0;
225 uint8_t i = 0;
226 while (i < 9 && ptr + i != end && ptr[i] & 0b10000000) {
227 ret <<= 7;
228 ret |= (int64_t)(ptr[i] & 0b01111111);
229 i++;
230 }

235 *val = ret;
